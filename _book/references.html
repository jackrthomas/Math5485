    { name: 'Brook Taylor', born: 1685, died: 1731, area: 'analysis',
      items: [ { text: "Taylor's theorem and Taylor polynomials: used throughout, e.g. for convergence of iterations and as Hermite interpolation at a single node", lectures: [1, 3, 4, 5, 6, 9] } ] },
    { name: 'James Stirling', born: 1692, died: 1770, area: 'analysis',
      items: [ { text: "Stirling's formula for n!, used to estimate the equispaced node polynomial", lectures: [6] } ] },
    { name: 'Leonhard Euler', born: 1707, died: 1783, area: 'errors',
      items: [ { text: "Euler's number e and the Euler–Mascheroni constant γ: first examples of approximation and rates of convergence", lectures: [1] } ] },
    { name: 'Johann Heinrich Lambert', born: 1728, died: 1777, area: 'nonlinear',
      items: [ { text: 'The Lambert W function, computed by fixed-point iteration', lectures: [3, 4] } ] },
    { name: 'Alexandre-Théophile Vandermonde', born: 1735, died: 1796, area: 'interp',
      items: [ { text: 'Vandermonde matrices and their (bad) conditioning', lectures: [3] } ] },
    { name: 'Joseph-Louis Lagrange', born: 1736, died: 1813, area: 'interp',
      items: [ { text: 'Lagrange interpolation and the Lagrange polynomials', lectures: [6, 7, 8] } ] },
    { name: 'Lorenzo Mascheroni', born: 1750, died: 1800, area: 'errors',
      items: [ { text: 'The Euler–Mascheroni constant γ', lectures: [1] } ] },
    { name: 'Bernard Bolzano', born: 1781, died: 1848, area: 'analysis',
      items: [ { text: 'The intermediate value theorem, behind the change of sign theorem and bisection', lectures: [3] } ] },
    { name: 'Augustin-Louis Cauchy', born: 1789, died: 1857, area: 'analysis',

  if (!root) return;
  const START = 1550, END = 2000;
  const pos = y => ((y - START) / (END - START) * 100) + '%';
  // surname = last word (keeps "de"/"von" style names short enough for phones); also the #who-<surname> link key
  const surname = p => p.name.split(' ').pop();
  const key = p => surname(p).toLowerCase().normalize('NFD').replace(/[^a-z]/g, '');
  const lectureLink = n => `<a class="lec" href="lectures/L${String(n).padStart(2, '0')}.html">Lecture ${n}</a>`;

  const axis = root.querySelector('.ht-axis'), rows = root.querySelector('.ht-rows');
  for (let y = START; y <= END; y += 50) {

  let filter = 'all', selected = null;

  function showCard(p) {
    const card = root.querySelector('.ht-card');
    if (!p) { card.innerHTML = '<span class="meta">Click on a name or bar for details.</span>'; card.style.removeProperty('--ht-accent'); return; }
    const a = AREAS[p.area];
    card.style.setProperty('--ht-accent', a.color);
    card.innerHTML = `<h4>${p.name}</h4>
      <div class="meta">${p.born}–${p.died} (aged about ${p.died - p.born}) · ${a.label}</div>
      <ul>${p.items.map(it => `<li>${it.text} <span class="text-muted">(${it.lectures.map(lectureLink).join(', ')})</span></li>`).join('')}</ul>`;
  }

  function render() {
    rows.innerHTML = '';
    // gridlines span the rows (inside the track area)
    const grid = document.createElement('div');
    grid.style.cssText = 'position:absolute;top:0;bottom:0;right:0;left:var(--ht-name-w);pointer-events:none;';
    for (let y = START; y <= END; y += 50) grid.insertAdjacentHTML('beforeend', `<div class="ht-gridline" style="left:${pos(y)}"></div>`);
    rows.appendChild(grid);

    PEOPLE.filter(p => filter === 'all' || p.area === filter)
      .sort((a, b) => a.born - b.born)
      .forEach(p => {
        const btn = document.createElement('button');
        btn.type = 'button';
        btn.className = 'ht-row' + (selected === p ? ' selected' : '');
        btn.setAttribute('aria-label', `${p.name}, ${p.born} to ${p.died}`);
        btn.innerHTML = `<span class="ht-name" title="${p.name}"><span class="full">${p.name}</span><span class="short">${surname(p)}</span></span>
          <span class="ht-track">
            <span class="ht-bar" style="left:${pos(p.born)};width:calc(${pos(p.died)} - ${pos(p.born)});background:${AREAS[p.area].color}"></span>
            ${(p.died - START) / (END - START) > 0.8
              ? `<span class="ht-years" style="right:calc(100% - ${pos(p.born)});padding:0 .3rem 0 0">${p.born}–${p.died}</span>`
              : `<span class="ht-years" style="left:${pos(p.died)}">${p.born}–${p.died}</span>`}
          </span>`;
        btn.addEventListener('click', () => { selected = p; render(); showCard(p); });
        rows.appendChild(btn);
      });
  }

  // filter chips
  const chips = root.querySelector('.ht-filters');
  const addChip = (key, label, color) => {
    const b = document.createElement('button');
    b.type = 'button'; b.className = 'ht-chip'; b.dataset.key = key;
    b.setAttribute('aria-pressed', key === filter);
    b.innerHTML = (color ? `<span class="dot" style="background:${color}"></span>` : '') + label;
    b.addEventListener('click', () => {
      filter = key;
      chips.querySelectorAll('.ht-chip').forEach(c => c.setAttribute('aria-pressed', c.dataset.key === key));
      if (selected && filter !== 'all' && selected.area !== filter) { selected = null; showCard(null); }
      render();
    });
    chips.appendChild(b);
  };
  addChip('all', 'All');
  Object.entries(AREAS).forEach(([k, a]) => addChip(k, a.label, a.color));

  // link straight to a person: references.html#who-newton
  function fromHash() {
    const m = location.hash.match(/^#who-([a-z]+)$/);
    const p = m && PEOPLE.find(q => key(q) === m[1]);
    if (!p) return false;
    filter = 'all';
    chips.querySelectorAll('.ht-chip').forEach(c => c.setAttribute('aria-pressed', c.dataset.key === 'all'));
    selected = p; render(); showCard(p);
    root.scrollIntoView({ block: 'start' });
    return true;
  }
  window.addEventListener('hashchange', fromHash);
  if (!fromHash()) { render(); showCard(null); }
})();
</script>
